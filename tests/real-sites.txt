# The browser on real HTTPS sites, through tools/test-real-sites.sh (which
# sets up DNS and a pass-through relay so this works from a sandbox whose
# only way out is a proxy). The firewall's built-in web policy lets the
# browser reach any host on port 443, so no dialogs come up.
wait 12
expect started terminal
expect [net] address 203.0.113.15
type run browser\n
wait 12
expect started browser
# TLS 1.3 with X25519.
type en.wikipedia.org\n
wait 25
expect [net] browser connected to en.wikipedia.org:443
shot 01-wikipedia-tls13
# TLS 1.3 with P-256 (bing.com has no X25519), then a redirect to www.
key ctrl-l
type bing.com\n
wait 45
expect [net] browser connected to bing.com:443
shot 02-bing-p256
# TLS 1.2 only, ECDHE-RSA with ChaCha20-Poly1305.
key ctrl-l
type www.eff.org\n
wait 25
expect [net] browser connected to www.eff.org:443
shot 03-eff-tls12-chacha
# TLS 1.2 only, ECDHE-RSA with AES-256-GCM-SHA384.
key ctrl-l
type www.mit.edu\n
wait 25
expect [net] browser connected to www.mit.edu:443
shot 04-mit-tls12-aes256
# TLS 1.2 with P-256 on samsung.com, which redirects to TLS 1.3 on www.
key ctrl-l
type samsung.com\n
wait 30
expect [net] browser connected to samsung.com:443
shot 05-samsung-tls12-then-13
key ctrl-l
type about:security\n
wait 4
shot 06-security
