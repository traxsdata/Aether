# Run with AETHER_DISK=none after tools/build-browser.sh built
# build/apps/browser.app: start the browser, show about:home, open
# example.com (no firewall question: the browser's web policy allows it),
# then about:security and the http warning.
# Without internet access on the host the page is an error page that shows
# webnet's message.

wait 12
expect started browser

type about:security\n
wait 4
shot 04-browser-security
key pgdn
wait 2

key ctrl-l
type http://example.com\n
